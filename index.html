    function exportCSV(counter) {

      const rows = [["Date", "Time", "ISO Timestamp", "Added", "Running Total"]];

      let running = 0;

      counter.entries.forEach(e => {

        running += e.delta;

        const d = new Date(e.timestamp);

        rows.push([

          d.toLocaleDateString(),

          d.toLocaleTimeString(),

          d.toISOString(),

          e.delta,

          running

        ]);

      });

      const csv = rows.map(r => r.map(v => `"${String(v).replace(/"/g, '""')}"`).join(",")).join("\n");

      const blob = new Blob([csv], { type: "text/csv;charset=utf-8;" });

      const url = URL.createObjectURL(blob);

      const a = document.createElement("a");

      a.href = url;

      a.download = `${counter.name.toLowerCase().replace(/[^a-z0-9]+/g, "-")}-data.csv`;

      a.click();

      URL.revokeObjectURL(url);

    }

    function renderSheetHtml(counter) {

      if (counter.entries.length === 0) {

        return `<div class="empty-state">No data yet. Add to this counter to record timestamps.</div>`;

      }


      let tableRows = "";

      let headerCols = "";


      if (state.sheetMode === "log") {

        headerCols = `

          <th>Date &amp; Time</th>

          <th class="num-col">Added</th>

          <th class="num-col">Total</th>

          <th></th>

        `;

        let running = 0;

        const enriched = counter.entries.map(e => {

          running += e.delta;

          return { ...e, running };

        }).reverse();


        tableRows = enriched.map(e => {

          const dt = formatDateTime(e.timestamp);

          return `

            <tr>

              <td>${dt.date} <span style="color:var(--muted)">${dt.time}</span></td>

              <td class="num-col">${e.delta > 0 ? "+" : ""}${e.delta}</td>

              <td class="num-col">${e.running}</td>

              <td class="num-col">

                <button class="del-entry" data-del-entry="${e.id}" title="Remove entry">&times;</button>

              </td>

            </tr>

          `;

        }).join("");

      } else {

        headerCols = `

          <th>Day</th>

          <th class="num-col">Entries</th>

          <th class="num-col">Day Total</th>

        `;

        const byDay = {};

        counter.entries.forEach(e => {

          const d = new Date(e.timestamp);

          const key = `${d.getFullYear()}-${String(d.getMonth() + 1).padStart(2, "0")}-${String(d.getDate()).padStart(2, "0")}`;

          if (!byDay[key]) {

            byDay[key] = {

              label: d.toLocaleDateString(undefined, { month: "short", day: "numeric", year: "numeric" }),

              count: 0,

              sum: 0

            };

          }

          byDay[key].count += 1;

          byDay[key].sum += e.delta;

        });


        tableRows = Object.keys(byDay).sort().reverse().map(k => {

          const day = byDay[k];

          return `

            <tr>

              <td>${day.label}</td>

              <td class="num-col">${day.count}</td>

              <td class="num-col" style="font-weight:600">${day.sum}</td>

            </tr>

          `;

        }).join("");

      }


      return `

        <div class="sheet-controls">

          <div class="pill-group">

            <button class="pill ${state.sheetMode === "log" ? "active" : ""}" data-sheet-mode="log">Timestamp Log</button>

            <button class="pill ${state.sheetMode === "daily" ? "active" : ""}" data-sheet-mode="daily">Daily Summary</button>

          </div>

          <button class="btn-ghost" id="export-csv-btn">Export CSV</button>

        </div>

        <div class="sheet-wrapper">

          <table>

            <thead><tr>${headerCols}</tr></thead>

            <tbody>${tableRows}</tbody>

          </table>

        </div>

      `;

    }


    function attachSheetListeners(app, counter) {

      app.querySelectorAll("[data-sheet-mode]").forEach(btn => {

          state.sheetMode = btn.dataset.sheetMode;

          render();

        });

      });


      const exportBtn = document.getElementById("export-csv-btn");

      if (exportBtn) {

        exportBtn.addEventListener("click", () => exportCSV(counter));

      }


      app.querySelectorAll("[data-del-entry]").forEach(btn => {

        btn.addEventListener("click", () => {

          const entryObj = counter.entries.find(e => e.id === btn.dataset.delEntry);

          deleteEntry(counter.id, entryObj);

        });

      });

    }
